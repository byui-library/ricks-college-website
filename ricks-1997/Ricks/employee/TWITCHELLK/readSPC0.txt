#include <fstream>

void main (void)
{
   const int SUBGROUP_COUNT (25);
   const int SAMPLE_COUNT (5);
   int dataValues[SUBGROUP_COUNT][SAMPLE_COUNT];

   // open the file containing the data values
   ifstream firstFile ("/home/twitchellk/examples/spc0.txt");
   for (int i = 0; i < SUBGROUP_COUNT; ++i)
      for (int j = 0; j < SAMPLE_COUNT; ++j)
         firstFile >> dataValues[i][j];
   firstFile.close (); // close the input file
}
