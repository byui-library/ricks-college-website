Adding complex numbers

/***********************************************************************
* Program:
*    Homework 0, Adding Complex Numbers
* Author:
*    Brother Twitchell, 7 July 1999
* Summary: 
*    Provides an example of a class and operator overloading.
*    Instantiates two Complex objects, adds them, and outputs the sum.
***********************************************************************/

#include <iostream>

class Complex
/***********************************************************************
* Each instantiation contains exactly one complex number. Both the
* real and imaginary parts of the complex number are floating-point
* values. For the complex number a+bi, a is represented by variable
* real and b is represented by variable imaginary.
***********************************************************************/
{
   private:
      float real;      // real part of the imaginary number
      float imaginary; // imaginary part of the imaginary number
   public:
      // the constructor has default values of 0.0 for the two parameters
      Complex (float suppliedReal = 0.0, float suppliedImaginary = 0.0);
      float getReal (void) const;      // returns the real portion
      float getImaginary (void) const; // returns the imaginary portion
};

Complex::Complex (float suppliedReal = 0.0, float suppliedImaginary = 0.0)
/***********************************************************************
* Initializes real and imaginary when a complex number object is
* instantiated. Receives the values intended for the real and imaginary
* portions of the complex number. Each of these has a value of 0.0 if
* no value is supplied. The received values are assigned to private
* data members real and imaginary.
***********************************************************************/
{
   // store supplied (or default) real and imaginary values
   real = suppliedReal;
   imaginary = suppliedImaginary;
}

float Complex::getReal (void) const
/***********************************************************************
* Returns the real portion of the complex number (i.e., a of a+bi)
* stored in private data member real.
***********************************************************************/
{
   return (real);
}

float Complex::getImaginary (void) const
/***********************************************************************
* Returns the imaginary portion of the complex number (i.e., b of a+bi)
* stored in private data member imaginary.
***********************************************************************/
{
   return (imaginary);
}

Complex operator + (const Complex &operand1, const Complex &operand2)
/***********************************************************************
* Receives two Complex objects and returns the sums of the values. The
* sum is obtained by adding the real portions of the two objects and by
* adding the imaginary portions of the two objects. The sum is returned
* as a Complex object.
***********************************************************************/
{
   // get the real values of the two complex objects and add them
   // get the imaginary values of the two complex objects and add them
   // use the summed real and imaginary values and instantiate a new
   //    complex object
   // return the instantiated complex object containing the sum
   return (Complex ((operand1.getReal () + operand2.getReal ()),
                    (operand1.getImaginary () + operand2.getImaginary ())));
}

ostream &operator << (ostream &outFile, const Complex &number)
/***********************************************************************
* Output to the received ostream object the received complex number,
* in the form of a+bi, where a is the real part and b is the imaginary
* part. The destination of the output (outFile) is returned so output
* can be chained
***********************************************************************/
{
   // output the complex number in the form of a+bi
   outFile << number.getReal () << "+" << number.getImaginary () << "i";
   return (outFile);
}

void main (void)
/***********************************************************************
* Instantiate two complex number objects, output the complex numbers
* and their sum. Demonstrates that class Complex works as intended.
***********************************************************************/
{
   Complex complexNumber1 (3.4, 7.2);
   Complex complexNumber2 (2.8, 19.0);

   // output the two complex numbers and their sum; by performing the
   //    addition in the cout statement, we demonstrate that the operator
   //    overloading of + correctly returns a Complex object
   cout << complexNumber1 << " + "
        << complexNumber2 << " = "
        << (complexNumber1 + complexNumber2) << endl;
}
