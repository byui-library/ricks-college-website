                COIN   WALK_THRU   GATE   GATE+   GATE.D (D input to GATE)

                0      1           0      1       1
                0      1           1      1       1

        /* Define Logic Operators */
       
            /* 
                AND = &

        /* Define Output Pin(s) */

            Pin 15 = GATE;
                      

        /* Define Input Pins */
        
            Pin 1  = CLK;   /* Clock Input */
            Pin 2  = WALK_THRU;
            Pin 3  = COIN;
            Pin 11 = !OE;   /* Output Enable Input -- Ground this pin. */


        /* Logic Equations */

            GATE.D = (GATE & !COIN) # (!GATE & WALK_THRU);
