Cubing an integer read from a file

/***********************************************************************
* Program:
*    Homework 0, Cube a Number
* Author:
*    Brother Twitchell, 2 July 1999
* Summary: 
*    This program reads a number from a text file, cubes the number,
*    and outputs the result.
***********************************************************************/

#include <iostream>
#include <fstream>

int cubedInteger (int *integerPointer)
/***********************************************************************
* Returns the cube of the supplied integer value.
* Receives a pointer to the value to be cubed.
* Returns the cube of the passed value.
***********************************************************************/
{
   // return the cube the supplied value
   return (*integerPointer * *integerPointer * *integerPointer);
}

void main (void)
/***********************************************************************
* Opens the input file and reads the number to be cubed. Outputs the
* original and cubed values. Closes the input file.
***********************************************************************/
{
   int integerValueFromInputFile (0);

   // open the input file, read a single integer from it, and close it
   ifstream inputFile ("number.txt");
   inputFile >> integerValueFromInputFile;
   inputFile.close ();

   // output the original value and its cube
   cout << "Impress your date! ";
   cout << "The cube of " << integerValueFromInputFile << " is "
        << cubedInteger (&integerValueFromInputFile) << "." << endl;
}
