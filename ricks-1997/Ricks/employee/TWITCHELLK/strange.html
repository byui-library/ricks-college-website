Array notation is commutative 

#include <iostream>

void main()
{
   int array[5] = {0, 1, 2, 3, 4};

   // each of the following is syntactically correct
   //  and produces the same output
   cout << array[3] << endl;
   cout << *(array + 3) << endl;
   cout << *(3 + array) << endl;
   cout << 3[array] << endl;
}

Order of output 

#include <iostream>

void main ()
{
   int i (0);
   int j (5);
   cout << i << " - " << (i = j * j) << " - " << i << endl;
   // output is 25 - 25 - 0
}

Swapping values without a third variable 

#include <iostream>

void main ()
{
   int i (2);
   int j (4);
   cout << "before: " << i << " = i, " << j << " = j" << endl;
   i = (j = j) + (j = i) - j;
   cout << " after: " << i << " = i, " << j << " = j" << endl;
   // output is before: 2 = i, 4 = j
   //            after: 4 = i, 2 = j
}

This is null

contributed by Kevin Henry 

#include <iostream>

class ThisIsNull
{
  public:
    void test (void);
};

void ThisIsNull::test (void)
{
  if (this == NULL)
    cout << "this is NULL!!\n"; // this is actually output
}

void main (void)
{
  ThisIsNull *nullPointer = NULL;
  nullPointer->test ();
}
